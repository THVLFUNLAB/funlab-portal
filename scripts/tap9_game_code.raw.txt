// --- Điểm mỗi câu đúng (10 câu × 5đ = tối đa 50đ, khớp giới hạn hệ thống) ---
const POINTS_PER_QUESTION = 5;
const MAX_SCORE = 50;
const PERFECT_TIME_LIMIT = 120; // giây — điều kiện huy hiệu "Thần Đèn Hóa Học"

function shuffleArray(arr) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

// Mô phỏng chiều cao cột khói (cm) — công thức minh họa trực quan, KHÔNG phải
// công thức động học chính xác tuyệt đối, chỉ nhằm thể hiện xu hướng đồng biến
// với nồng độ H2O2 và khối lượng KMnO4 (đúng với nội dung kiến thức đã dạy).
function computeSmokeHeight(concentration, mass) {
  const raw = concentration * 1.2 + mass * 18;
  return Math.max(8, Math.min(150, Math.round(raw)));
}

function Game({ onGameComplete }) {
  const [phase, setPhase] = useState('intro'); // intro -> lab -> quiz -> result
  const [level, setLevel] = useState('THCS');
  const [concentration, setConcentration] = useState(15);
  const [mass, setMass] = useState(2);
  const [smokeHeight, setSmokeHeight] = useState(0);
  const [isReacting, setIsReacting] = useState(false);
  const [hasTriggeredOnce, setHasTriggeredOnce] = useState(false);

  const [questions, setQuestions] = useState([]);
  const [qIndex, setQIndex] = useState(0);
  const [selected, setSelected] = useState(null);
  const [isChecked, setIsChecked] = useState(false);
  const [score, setScore] = useState(0);
  const [answersLog, setAnswersLog] = useState([]);

  const [startedAt, setStartedAt] = useState(null);
  const [elapsed, setElapsed] = useState(0);

  useEffect(() => {
    if (!startedAt) return;
    const timer = setInterval(() => {
      setElapsed(Math.floor((Date.now() - startedAt) / 1000));
    }, 1000);
    return () => clearInterval(timer);
  }, [startedAt]);

  const handleStart = (chosenLevel) => {
    setLevel(chosenLevel);
    const pool = QUESTION_BANK.filter((q) => q.cap === chosenLevel);
    setQuestions(shuffleArray(pool));
    setStartedAt(Date.now());
    setPhase('lab');
  };

  const handleTriggerReaction = () => {
    if (isReacting) return;
    setIsReacting(true);
    setSmokeHeight(0);
    setTimeout(() => {
      setSmokeHeight(computeSmokeHeight(concentration, mass));
      setHasTriggeredOnce(true);
      setIsReacting(false);
    }, 900);
  };

  const handleGoToQuiz = () => {
    setPhase('quiz');
    setQIndex(0);
    setSelected(null);
    setIsChecked(false);
  };

  const currentQuestion = questions[qIndex];

  const handleSelectOption = (optIndex) => {
    if (isChecked) return;
    setSelected(optIndex);
  };

  const handleCheckAnswer = () => {
    if (selected === null || !currentQuestion) return;
    const isCorrect = selected === currentQuestion.ans;
    if (isCorrect) setScore((s) => s + POINTS_PER_QUESTION);
    setAnswersLog((log) => [...log, { qId: currentQuestion.id, isCorrect, level: currentQuestion.level, safety: currentQuestion.safety }]);
    setIsChecked(true);
  };

  const handleNextQuestion = () => {
    if (qIndex + 1 < questions.length) {
      setQIndex((i) => i + 1);
      setSelected(null);
      setIsChecked(false);
    } else {
      setPhase('result');
    }
  };

  const finalTimeInSeconds = elapsed;
  const safetyQuestionsInSet = questions.filter((q) => q.safety);
  const safetyAllCorrect =
    safetyQuestionsInSet.length > 0 &&
    safetyQuestionsInSet.every((sq) => answersLog.find((a) => a.qId === sq.id)?.isCorrect);
  const isPerfect = score === MAX_SCORE && finalTimeInSeconds <= PERFECT_TIME_LIMIT;

  const badges = [];
  if (isPerfect) badges.push({ id: 'than-den', label: 'Thần Đèn Hóa Học', icon: '🧞' });
  if (safetyAllCorrect) badges.push({ id: 'an-toan', label: 'Bậc Thầy An Toàn', icon: '🛡️' });

  const handleSubmit = () => {
    onGameComplete({
      score,
      timeInSeconds: finalTimeInSeconds,
      level,
      answersLog,
    });
    setPhase('submitted');
  };

  // ============================== INTRO ==============================
  if (phase === 'intro') {
    return (
      <div style={{ minHeight: 480, background: 'linear-gradient(180deg,#0f172a,#020617)', borderRadius: 24, padding: 32, color: '#e2e8f0', fontFamily: 'sans-serif', position: 'relative', overflow: 'hidden' }}>
        <div style={{ position: 'absolute', top: -80, right: -80, width: 240, height: 240, borderRadius: '50%', background: 'radial-gradient(circle, rgba(34,211,238,0.18), transparent 70%)' }} />
        <div style={{ display: 'flex', alignItems: 'center', gap: 10, marginBottom: 8 }}>
          <Sparkles size={22} color="#22d3ee" />
          <span style={{ fontSize: 12, letterSpacing: 2, fontWeight: 800, color: '#22d3ee', textTransform: 'uppercase' }}>Tập 9 · FunLab Challenge</span>
        </div>
        <h1 style={{ fontSize: 28, fontWeight: 900, margin: '4px 0 6px', lineHeight: 1.2 }}>🧞 THẦN ĐÈN KHÓI TRẮNG</h1>
        <p style={{ color: '#94a3b8', margin: '0 0 4px', fontSize: 14 }}>Phản ứng phân hủy tỏa nhiệt: H₂O₂ + KMnO₄</p>
        <p style={{ color: '#67e8f9', fontStyle: 'italic', margin: '4px 0 20px', fontSize: 13 }}>"Hãy làm cho khoa học không chỉ để hiểu, mà còn để ngắm nhìn!"</p>

        <div style={{ background: 'rgba(239,68,68,0.08)', border: '1px solid rgba(239,68,68,0.35)', borderRadius: 16, padding: 16, marginBottom: 20, display: 'flex', gap: 12 }}>
          <ShieldCheck size={22} color="#f87171" style={{ flexShrink: 0, marginTop: 2 }} />
          <div style={{ fontSize: 13, color: '#fecaca', lineHeight: 1.6 }}>
            <strong style={{ color: '#fca5a5' }}>An toàn thí nghiệm thật:</strong> luôn đeo kính bảo hộ, găng tay nitrile và dùng bình thủy tinh chịu nhiệt Borosilicate. Trò chơi mô phỏng dưới đây hoàn toàn an toàn để khám phá.
          </div>
        </div>

        <p style={{ fontSize: 13, color: '#cbd5e1', marginBottom: 12 }}>Chọn cấp học để bắt đầu hành trình:</p>
        <div style={{ display: 'flex', gap: 14 }}>
          <button onClick={() => handleStart('THCS')} style={{ flex: 1, padding: '18px 16px', borderRadius: 16, border: '1px solid rgba(34,211,238,0.4)', background: 'rgba(34,211,238,0.08)', color: '#e2e8f0', cursor: 'pointer', textAlign: 'left' }}>
            <div style={{ fontWeight: 800, fontSize: 15, marginBottom: 4 }}>🔬 THCS</div>
            <div style={{ fontSize: 12, color: '#94a3b8' }}>Hiện tượng, tính chất, ứng dụng đời sống</div>
          </button>
          <button onClick={() => handleStart('THPT')} style={{ flex: 1, padding: '18px 16px', borderRadius: 16, border: '1px solid rgba(168,85,247,0.4)', background: 'rgba(168,85,247,0.08)', color: '#e2e8f0', cursor: 'pointer', textAlign: 'left' }}>
            <div style={{ fontWeight: 800, fontSize: 15, marginBottom: 4 }}>⚗️ THPT</div>
            <div style={{ fontSize: 12, color: '#94a3b8' }}>Số oxi hóa, cân bằng electron, tính toán mol</div>
          </button>
        </div>
      </div>
    );
  }

  // ============================== LAB (SIMULATION) ==============================
  if (phase === 'lab') {
    const barHeightPx = Math.round((smokeHeight / 150) * 220);
    return (
      <div style={{ minHeight: 480, background: 'linear-gradient(180deg,#0f172a,#020617)', borderRadius: 24, padding: 28, color: '#e2e8f0', fontFamily: 'sans-serif' }}>
        <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', marginBottom: 18 }}>
          <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
            <Beaker size={20} color="#22d3ee" />
            <span style={{ fontWeight: 800, fontSize: 14 }}>Phòng Thí Nghiệm Ảo</span>
          </div>
          <div style={{ fontSize: 11, padding: '4px 10px', borderRadius: 999, background: 'rgba(148,163,184,0.15)', color: '#cbd5e1' }}>
            <Clock size={12} style={{ verticalAlign: -2, marginRight: 4 }} />{elapsed}s
          </div>
        </div>

        <div style={{ display: 'flex', gap: 24, flexWrap: 'wrap' }}>
          <div style={{ flex: '1 1 220px', display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'flex-end', background: 'rgba(15,23,42,0.6)', borderRadius: 16, padding: 16, minHeight: 260, position: 'relative' }}>
            <div style={{
              width: 46, height: Math.max(20, barHeightPx), borderRadius: '10px 10px 4px 4px',
              background: 'linear-gradient(180deg, rgba(255,255,255,0.9), rgba(203,213,225,0.4))',
              boxShadow: '0 0 24px rgba(255,255,255,0.5)',
              transition: 'height 0.9s ease-out', marginBottom: 6,
            }} />
            <div style={{ width: 64, height: 34, borderRadius: '4px 4px 12px 12px', background: 'rgba(103,232,249,0.25)', border: '2px solid rgba(103,232,249,0.5)' }} />
            <p style={{ marginTop: 10, fontSize: 12, color: '#94a3b8' }}>
              {isReacting ? 'Đang phản ứng... 💨' : hasTriggeredOnce ? `Cột khói: ${smokeHeight} cm` : 'Chỉnh thông số rồi bấm "Châm phản ứng"'}
            </p>
          </div>

          <div style={{ flex: '1 1 260px' }}>
            <div style={{ marginBottom: 18 }}>
              <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: 12, marginBottom: 6 }}>
                <span style={{ color: '#94a3b8' }}>Nồng độ H₂O₂</span>
                <span style={{ fontWeight: 700, color: '#22d3ee' }}>{concentration}%</span>
              </div>
              <input type="range" min="3" max="35" value={concentration}
                onChange={(e) => setConcentration(Number(e.target.value))}
                style={{ width: '100%' }} />
            </div>
            <div style={{ marginBottom: 18 }}>
              <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: 12, marginBottom: 6 }}>
                <span style={{ color: '#94a3b8' }}>Khối lượng KMnO₄</span>
                <span style={{ fontWeight: 700, color: '#a855f7' }}>{mass} g</span>
              </div>
              <input type="range" min="0.5" max="5" step="0.5" value={mass}
                onChange={(e) => setMass(Number(e.target.value))}
                style={{ width: '100%' }} />
            </div>

            <button onClick={handleTriggerReaction} disabled={isReacting} style={{
              width: '100%', padding: '12px 16px', borderRadius: 14, border: 'none',
              background: isReacting ? '#334155' : 'linear-gradient(90deg,#f97316,#ef4444)',
              color: '#fff', fontWeight: 800, fontSize: 14, cursor: isReacting ? 'default' : 'pointer',
              display: 'flex', alignItems: 'center', justifyContent: 'center', gap: 8,
            }}>
              <Flame size={16} /> {isReacting ? 'Đang châm...' : 'Châm phản ứng'}
            </button>

            {hasTriggeredOnce && (
              <button onClick={handleGoToQuiz} style={{
                width: '100%', marginTop: 12, padding: '12px 16px', borderRadius: 14,
                border: '1px solid rgba(34,211,238,0.5)', background: 'rgba(34,211,238,0.1)',
                color: '#67e8f9', fontWeight: 800, fontSize: 14, cursor: 'pointer',
                display: 'flex', alignItems: 'center', justifyContent: 'center', gap: 8,
              }}>
                Vào phần câu hỏi <ArrowRight size={16} />
              </button>
            )}
          </div>
        </div>
      </div>
    );
  }

  // ============================== QUIZ ==============================
  if (phase === 'quiz' && currentQuestion) {
    return (
      <div style={{ minHeight: 480, background: 'linear-gradient(180deg,#0f172a,#020617)', borderRadius: 24, padding: 28, color: '#e2e8f0', fontFamily: 'sans-serif' }}>
        <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: 16 }}>
          <span style={{ fontSize: 11, fontWeight: 800, color: '#94a3b8', letterSpacing: 1 }}>
            CÂU {qIndex + 1}/{questions.length} · {currentQuestion.level.toUpperCase()}
          </span>
          <div style={{ display: 'flex', alignItems: 'center', gap: 10, fontSize: 12 }}>
            <span style={{ color: '#fbbf24', fontWeight: 800 }}><Zap size={12} style={{ verticalAlign: -2 }} /> {score} đ</span>
            <span style={{ color: '#94a3b8' }}><Clock size={12} style={{ verticalAlign: -2 }} /> {elapsed}s</span>
          </div>
        </div>

        <div style={{ width: '100%', height: 6, borderRadius: 999, background: 'rgba(148,163,184,0.15)', marginBottom: 20, overflow: 'hidden' }}>
          <div style={{ width: `${((qIndex + (isChecked ? 1 : 0)) / questions.length) * 100}%`, height: '100%', background: '#22d3ee', transition: 'width 0.4s' }} />
        </div>

        <p style={{ fontSize: 16, fontWeight: 700, lineHeight: 1.5, marginBottom: 18 }}>{currentQuestion.q}</p>

        <div style={{ display: 'flex', flexDirection: 'column', gap: 10, marginBottom: 18 }}>
          {currentQuestion.options.map((opt, idx) => {
            const isSelected = selected === idx;
            const isCorrectOpt = idx === currentQuestion.ans;
            let border = 'rgba(148,163,184,0.25)';
            let bg = 'rgba(15,23,42,0.5)';
            if (isChecked && isCorrectOpt) { border = '#22c55e'; bg = 'rgba(34,197,94,0.12)'; }
            else if (isChecked && isSelected && !isCorrectOpt) { border = '#ef4444'; bg = 'rgba(239,68,68,0.12)'; }
            else if (!isChecked && isSelected) { border = '#22d3ee'; bg = 'rgba(34,211,238,0.1)'; }
            return (
              <button key={idx} onClick={() => handleSelectOption(idx)} disabled={isChecked} style={{
                textAlign: 'left', padding: '12px 14px', borderRadius: 12, border: `1.5px solid ${border}`,
                background: bg, color: '#e2e8f0', fontSize: 13.5, cursor: isChecked ? 'default' : 'pointer',
                display: 'flex', alignItems: 'center', gap: 10,
              }}>
                {isChecked && isCorrectOpt && <CheckCircle2 size={16} color="#22c55e" />}
                {isChecked && isSelected && !isCorrectOpt && <XCircle size={16} color="#ef4444" />}
                <span>{opt}</span>
              </button>
            );
          })}
        </div>

        {isChecked && (
          <div style={{ background: 'rgba(34,211,238,0.06)', border: '1px solid rgba(34,211,238,0.25)', borderRadius: 12, padding: 14, marginBottom: 18, fontSize: 12.5, color: '#cbd5e1', lineHeight: 1.6 }}>
            <strong style={{ color: '#67e8f9' }}>Giải thích: </strong>{currentQuestion.explain}
          </div>
        )}

        {!isChecked ? (
          <button onClick={handleCheckAnswer} disabled={selected === null} style={{
            width: '100%', padding: '12px 16px', borderRadius: 14, border: 'none',
            background: selected === null ? '#334155' : 'linear-gradient(90deg,#22d3ee,#0891b2)',
            color: '#fff', fontWeight: 800, fontSize: 14, cursor: selected === null ? 'default' : 'pointer',
          }}>Kiểm tra đáp án</button>
        ) : (
          <button onClick={handleNextQuestion} style={{
            width: '100%', padding: '12px 16px', borderRadius: 14, border: 'none',
            background: 'linear-gradient(90deg,#a855f7,#7c3aed)', color: '#fff', fontWeight: 800, fontSize: 14, cursor: 'pointer',
            display: 'flex', alignItems: 'center', justifyContent: 'center', gap: 8,
          }}>
            {qIndex + 1 < questions.length ? 'Câu tiếp theo' : 'Xem kết quả'} <ArrowRight size={16} />
          </button>
        )}
      </div>
    );
  }

  // ============================== RESULT ==============================
  if (phase === 'result' || phase === 'submitted') {
    return (
      <div style={{ minHeight: 480, background: 'linear-gradient(180deg,#0f172a,#020617)', borderRadius: 24, padding: 32, color: '#e2e8f0', fontFamily: 'sans-serif', textAlign: 'center' }}>
        <PartyPopper size={40} color="#fbbf24" style={{ marginBottom: 10 }} />
        <h2 style={{ fontSize: 22, fontWeight: 900, marginBottom: 6 }}>Hoàn thành Tập 9!</h2>
        <p style={{ color: '#94a3b8', fontSize: 13, marginBottom: 22 }}>Cấp {level} · {elapsed} giây</p>

        <div style={{ display: 'flex', justifyContent: 'center', gap: 16, marginBottom: 22, flexWrap: 'wrap' }}>
          <div style={{ background: 'rgba(251,191,36,0.08)', border: '1px solid rgba(251,191,36,0.3)', borderRadius: 16, padding: '16px 24px' }}>
            <div style={{ fontSize: 28, fontWeight: 900, color: '#fbbf24' }}>{score}/{MAX_SCORE}</div>
            <div style={{ fontSize: 11, color: '#94a3b8' }}>Điểm số</div>
          </div>
          <div style={{ background: 'rgba(34,211,238,0.08)', border: '1px solid rgba(34,211,238,0.3)', borderRadius: 16, padding: '16px 24px' }}>
            <div style={{ fontSize: 28, fontWeight: 900, color: '#22d3ee' }}>{elapsed}s</div>
            <div style={{ fontSize: 11, color: '#94a3b8' }}>Thời gian</div>
          </div>
        </div>

        {badges.length > 0 && (
          <div style={{ marginBottom: 24 }}>
            <p style={{ fontSize: 12, color: '#94a3b8', marginBottom: 10 }}>🎉 Huy hiệu đạt được trong lượt chơi này:</p>
            <div style={{ display: 'flex', justifyContent: 'center', gap: 12, flexWrap: 'wrap' }}>
              {badges.map((b) => (
                <div key={b.id} style={{ background: 'rgba(168,85,247,0.1)', border: '1px solid rgba(168,85,247,0.4)', borderRadius: 14, padding: '10px 16px', fontSize: 13, fontWeight: 700 }}>
                  <span style={{ marginRight: 6 }}>{b.icon}</span>{b.label}
                </div>
              ))}
            </div>
          </div>
        )}

        {phase === 'result' ? (
          <button onClick={handleSubmit} style={{
            padding: '14px 32px', borderRadius: 16, border: 'none',
            background: 'linear-gradient(90deg,#22d3ee,#a855f7)', color: '#fff',
            fontWeight: 900, fontSize: 15, cursor: 'pointer',
            display: 'inline-flex', alignItems: 'center', gap: 8,
          }}>
            <Trophy size={18} /> Nộp kết quả về Bảng Vàng
          </button>
        ) : (
          <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'center', gap: 8, color: '#22c55e', fontWeight: 700, fontSize: 14 }}>
            <CheckCircle2 size={18} /> Đã nộp điểm thành công!
          </div>
        )}
      </div>
    );
  }

  return null;
}

return Game;
